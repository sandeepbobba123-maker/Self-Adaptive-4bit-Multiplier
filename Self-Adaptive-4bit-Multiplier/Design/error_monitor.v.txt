module error_monitor (
    input  wire [7:0] Exact_Product,
    input  wire [7:0] Approx_Product,
    input  wire [3:0] Error_Threshold,

    output wire [7:0] Absolute_Error,
    output wire Result_Pass
);

    wire [7:0] threshold_extended;

    assign threshold_extended =
        {4'b0000, Error_Threshold};

    assign Absolute_Error =
        (Exact_Product >= Approx_Product) ?
        (Exact_Product - Approx_Product) :
        (Approx_Product - Exact_Product);

    assign Result_Pass =
        (Absolute_Error <= threshold_extended);

endmodule
