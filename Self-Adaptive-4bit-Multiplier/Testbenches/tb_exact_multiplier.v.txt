`timescale 1ns / 1ps

module tb_exact_multiplier;

    reg [3:0] A, B;
    wire [7:0] Product;

    integer i, j;
    integer test_count;

    exact_multiplier_4bit DUT (
        .A(A),
        .B(B),
        .Product(Product)
    );

    initial begin

        test_count = 0;

        for (i = 0; i < 16; i = i + 1) begin
            for (j = 0; j < 16; j = j + 1) begin

                A = i;
                B = j;

                #10;

                if (Product !== (i * j))
                    $fatal(1, "FAILED: A=%d B=%d Product=%d",
                           A, B, Product);

                test_count = test_count + 1;

            end
        end

        $display("--------------------------------");
        $display("TOTAL TEST CASES = %d", test_count);
        $display("PASS: All 256 test cases successful");
        $display("--------------------------------");

        $finish;

    end

endmodule
