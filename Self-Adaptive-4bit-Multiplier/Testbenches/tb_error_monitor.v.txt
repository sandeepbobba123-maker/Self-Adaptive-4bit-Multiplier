`timescale 1ns / 1ps

module tb_error_monitor;

    reg [3:0] A, B;
    reg [3:0] Error_Threshold;

    wire [7:0] Exact_Product;
    wire [7:0] Approx_Product;
    wire [7:0] Absolute_Error;

    wire Result_Pass;

    exact_multiplier_4bit U1 (
        .A(A),
        .B(B),
        .Product(Exact_Product)
    );

    approx_multiplier_4bit U2 (
        .A(A),
        .B(B),
        .Product(Approx_Product)
    );

    error_monitor U3 (
        .Exact_Product(Exact_Product),
        .Approx_Product(Approx_Product),
        .Error_Threshold(Error_Threshold),
        .Absolute_Error(Absolute_Error),
        .Result_Pass(Result_Pass)
    );

    initial begin

        // TEST 1: Error equals threshold
        A = 7;
        B = 5;
        Error_Threshold = 7;

        #10;

        $display("TEST 1: Exact=%d Approx=%d Error=%d PASS=%b",
                 Exact_Product, Approx_Product,
                 Absolute_Error, Result_Pass);

        if (Exact_Product !== 35 ||
            Approx_Product !== 28 ||
            Absolute_Error !== 7 ||
            Result_Pass !== 1'b1)
            $fatal(1, "TEST 1 FAILED");

        $display("TEST 1 PASSED");

        // TEST 2: Error exceeds threshold
        A = 7;
        B = 5;
        Error_Threshold = 6;

        #10;

        $display("TEST 2: Exact=%d Approx=%d Error=%d PASS=%b",
                 Exact_Product, Approx_Product,
                 Absolute_Error, Result_Pass);

        if (Exact_Product !== 35 ||
            Approx_Product !== 28 ||
            Absolute_Error !== 7 ||
            Result_Pass !== 1'b0)
            $fatal(1, "TEST 2 FAILED");

        $display("TEST 2 PASSED");

        // TEST 3: Zero error
        A = 7;
        B = 4;
        Error_Threshold = 0;

        #10;

        $display("TEST 3: Exact=%d Approx=%d Error=%d PASS=%b",
                 Exact_Product, Approx_Product,
                 Absolute_Error, Result_Pass);

        if (Exact_Product !== 28 ||
            Approx_Product !== 28 ||
            Absolute_Error !== 0 ||
            Result_Pass !== 1'b1)
            $fatal(1, "TEST 3 FAILED");

        $display("TEST 3 PASSED");

        $display("--------------------------------");
        $display("ALL 3 TESTS PASSED SUCCESSFULLY");
        $display("--------------------------------");

        $finish;

    end

endmodule
